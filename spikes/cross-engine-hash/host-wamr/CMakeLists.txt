cmake_minimum_required(VERSION 3.14)
project(xhash_wamr C)

# -DWAMR_ROOT=/path/to/wasm-micro-runtime (checked out at the tag pinned in ../versions.env)
if(NOT WAMR_ROOT)
  message(FATAL_ERROR "set -DWAMR_ROOT")
endif()

set(WAMR_BUILD_PLATFORM linux)
set(WAMR_BUILD_TARGET X86_64)
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 1)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 0)
set(WAMR_BUILD_LIBC_WASI 1)
set(WAMR_BUILD_SIMD 1)
set(WAMR_BUILD_BULK_MEMORY 1)
# Deliberately permissive so counterexamples (atomics) are accepted if the engine can run them.
set(WAMR_BUILD_SHARED_MEMORY 1)
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_MULTI_MODULE 0)

include(${WAMR_ROOT}/build-scripts/runtime_lib.cmake)
add_executable(xhash-wamr main.c ${WAMR_RUNTIME_LIB_SOURCE})
target_include_directories(xhash-wamr PRIVATE ${WAMR_ROOT}/core/iwasm/include)
target_link_libraries(xhash-wamr m dl pthread)
